# The differential oracle's corpus — CEL source and the frozen answer.
#
# This file is DATA. Nothing in it is Rust and nothing in it is CEL beyond the
# `expr:` line itself; the `want:` notation is a canonical rendering defined by
# `render()`/`render_error()` in tests/oracle.rs and by nothing else. That is
# deliberate: an oracle written as a second evaluator gets rewritten by the same
# commit that rewrites the subject, and the gate disappears without anyone
# noticing. A text file does not.
#
# ---------------------------------------------------------------------------
# Record format
# ---------------------------------------------------------------------------
#
#   expr: <CEL source, one line>
#   feat: <space separated tags from REQUIRED_COVERAGE>
#   want: <canonical rendering of the expected Value, or error(...)>
#   cfg:  <optional cargo feature this case needs>
#
# Records are separated by a blank line. `#` starts a comment line.
#
# ---------------------------------------------------------------------------
# The bindings every case sees (built by `fixed_context()` in oracle.rs)
# ---------------------------------------------------------------------------
#
#   i        int    42
#   neg      int    -7
#   u        uint   7
#   d        double 1.5
#   b        bool   true
#   s        string "hello"
#   by       bytes  b"hi"
#   nil      null
#   xs       list   [1, 2, 3, 4]
#   strs     list   ["a", "bb", "ccc"]
#   empty    list   []
#   m        map    {"a": 1, "b": 2}
#   nested   map    {"inner": {"k": 5}}
#   people   list   [{"name": "ann", "age": 30}, {"name": "bob", "age": 20}]
#   opt_some optional(9)
#   opt_none optional.none
#
#   double_it(int) -> int      host function, doubles its argument
#   boom(int) -> error         host function that always fails
#

# `&&` and `||` are commutative over errors: an error on either side is
# absorbed when the *other* side decides the result. The six cases above only
# ever put the error on the right, where plain short-circuiting already skips
# it, so none of them can tell absorption apart from short-circuiting. These
# put it on the left, where the operator has to evaluate both sides and then
# discard a real error.

expr: undefined_name && false
feat: logic_and error
want: bool(false)

expr: undefined_name && true
feat: logic_and error
want: error(UndeclaredReference: undefined_name)

expr: undefined_name || true
feat: logic_or error
want: bool(true)

expr: undefined_name || false
feat: logic_or error
want: error(UndeclaredReference: undefined_name)

expr: undefined_name && undefined_name
feat: logic_and error
want: error(UndeclaredReference: undefined_name)

# The absorbed error is raised deep inside the left operand rather than by
# resolving a name, so recognising it cannot be a property of the operand's
# shape.

expr: (1 / 0 == 1) && false
feat: logic_and arith error
want: bool(false)

# The error lands while a list literal is half-built, so absorbing it has to
# discard the partial aggregate as well as the error.

expr: ([1, 2, undefined_name][0] == 1) && false
feat: logic_and literal_list index_list error
want: bool(false)

# A non-bool operand is an overload failure, and on the *left* that failure is
# absorbed exactly like any other error -- `1 && false` is `false`. So "is this
# a bool" is not a precondition the operator can check before deciding; it is
# one more outcome the merge weighs against the other side.

want: error(NoSuchOverload)

# A member call on an identifier receiver is compiled as a namespaced probe
# followed by the receiver path, and the probe pops the arguments to ask about
# them. On a miss they are handed straight to the receiver path rather than
# pushed back, so what is popped once is never popped twice.
#
# These two pin the hand-off against the case where it never arrives: the
# receiver load itself raises, a short-circuit absorbs it, and the arguments
# must not still be waiting when the OTHER operand makes its own member call.
# `"..."` receivers are literals, which is the arm that pops its own arguments,
# so a surviving hand-off is exactly what they would read instead. Both answers
# below are the ones the walker gives, which never moves an argument twice.
expr: undefined_name.startsWith("zzz") || "hello world".contains("o w")
feat: string_fn logic_or error
want: bool(true)

expr: undefined_name.startsWith("o w") && "hello world".contains("zzz")
feat: string_fn logic_and error
want: bool(false)

# ---------------------------------------------------------------------------
# A comprehension variable shadows the package namespace
# ---------------------------------------------------------------------------
#
# Decided by cel-spec, doc/langdef.md (name resolution), not by the incumbent:
# "in a comprehension like `[1].exists(x, x == 1)`, `x` is a local variable
# which shadows any identifier named `x` in ancestor scopes or the package
# namespace". So a receiver a macro bound is a VALUE, never a namespace, and
# `optional.of(...)` inside `xs.all(optional, ...)` is a member call `of` on the
# element -- which no overload accepts.
#
# The tree walker used to probe the namespace unconditionally and answer
# `bool(true)` here; the bytecode VM never did. The walker was changed to the
# spec's answer, so this row records a behaviour change, not a preserved one.

feat: macro_all optional error
want: error(UndeclaredReference: of)
